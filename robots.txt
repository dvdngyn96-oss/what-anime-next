# whatanimeshouldiwatchnext

User-agent: *
Allow: /

# Cloudflare Pages serves the whole repository root, so the build scripts and
# the test suite are publicly fetchable. They hold no secrets — the two
# credential files are gitignored and were never committed — but they are
# source, not content, and have no business in an index.
Disallow: /test/
Disallow: /build-catalogue.mjs
Disallow: /add-anilist-tags.mjs
Disallow: /backfill-genres.mjs
Disallow: /patch-studios.mjs
Disallow: /add-one.mjs
Disallow: /build-seo-pages.mjs
Disallow: /package.json
Disallow: /package-lock.json
Disallow: /make-og-image.html
Disallow: /schema.sql

# The vote endpoints. Nothing here is a document — they answer JSON to the
# page's own fetches — so there is nothing for a crawler to index, and a
# crawler walking them would spend the daily request allowance on nobody.
Disallow: /api/

# privacy.html is deliberately NOT disallowed. It is content, it is the only
# other real document on the site, and a page describing what is collected is
# no use to anyone if it cannot be found.

# app.js, styles.css and anime.json are deliberately NOT disallowed. The page
# renders nothing without them, so blocking them would leave a crawler holding
# an empty shell and judging the site on it.
